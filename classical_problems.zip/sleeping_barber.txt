    while(TRUE){
        down(&customer);
        down(mutex);
        waiting-=1;
        up(&barber);
        up(mutex);
        cut_hair;
    }
}

    down(mutex);
    if(waiting<CHAIRS){
        waiting+=1;
        up(customer);

        down(&mutex);
    }
}
